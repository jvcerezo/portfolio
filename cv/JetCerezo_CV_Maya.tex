%---------- Section
% \hrule draws in vertical mode, so the label and its underline sit tight
% together. \rule would be typeset into a line box and inherit a full
% \baselineskip of blank space above it.
\newcommand{\cvsection}[1]{%
  \par\addvspace{9pt}%
  \noindent{\small\bfseries\color{ink}\MakeUppercase{#1}}%
  \par\nobreak\vspace{2.5pt}%
  {\color{ink}\hrule height 0.4pt}%
  \nobreak\vspace{3pt}%
}

%---------- Entry: title, subtitle, date, right-side (optional location)
% tabular* with \extracolsep{\fill} rather than tabularx: an X column is a
% paragraph box whose first baseline is pushed down by a strut, which both
% opened a ~20pt gap under the organisation name and left the right-hand
% column sitting on a different baseline from the left. Plain l/r columns
% share one baseline, so the pairs line up and \\[-2pt] tunes the leading.
\newcommand{\entry}[4]{%
  \par\smallskip
  \noindent\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}l r@{}}
    \textbf{#1} & {\small\itshape\color{muted}#3} \\[-2pt]
    {\small\itshape\color{muted}#2} & {\small\itshape\color{muted}#4} \\
  \end{tabular*}\par\vspace{-2pt}
}

%---------- Project entry: title, date
